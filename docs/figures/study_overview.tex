\newcommand{\cube}[4]{% name, position, colour, text
  \node[rectangle, draw=#3, line width=1pt, fill=white, align=center,
        text width=30mm, minimum height=11mm] (#1) at #2 {#4};
  \draw[#3, line width=1pt, fill=#3!12]
    (#1.north west) -- ++(2.5mm,2.5mm) -- ($(#1.north east)+(2.5mm,2.5mm)$)
    -- (#1.north east) -- cycle;
  \draw[#3, line width=1pt, fill=#3!25]
    (#1.north east) -- ++(2.5mm,2.5mm) -- ($(#1.south east)+(2.5mm,2.5mm)$)
    -- (#1.south east) -- cycle;
}
\cube{gen}{(0,0)}{inC}{\textbf{Simulated trials}\\signal $+$ noise}
\node[par, left=6mm of gen] (p1) {$A,\ \omega_A$};
\draw[arr, gray!70] (p1) -- (gen);
\begin{scope}[shift={($(gen.east)+(7mm,-4mm)$)}, x=1.6mm, y=7mm]
  \draw[gray!60] (0,0) -- (10,0);
  \draw[inC, thick, domain=0:10, samples=40] plot (\x, {exp(-(\x-5)^2/2.5)});
  \node[font=\tiny, gray] at (5,-0.35) {time};
\end{scope}
\node[proc, below=of gen] (avg) {Downsample to 100 Hz,\\average over trials};
\node[dat,  below=of avg, text width=36mm] (means) {Participant averages\\61 electrodes $\times$ 51 times};
\begin{scope}[shift={($(means.east)+(4mm,-4mm)$)}]
  \foreach \i in {0,...,9} \foreach \j in {0,...,5} {
    \pgfmathsetmacro{\v}{100*exp(-((\i-4.5)^2)/5 - ((\j-2.5)^2)/3)}
    \fill[inC!\v!white] (\i*0.8mm,\j*1.3mm) rectangle ++(0.8mm,1.3mm);}
  \draw[gray!60] (0,0) rectangle (8mm,7.8mm);
  \node[font=\tiny, gray, below] at (4mm,0) {time};
  \node[font=\tiny, gray, rotate=90, above] at (0,3.9mm) {electrode};
\end{scope}
\node[proc, draw=bgamC, line width=1pt, below=13mm of means, xshift=-27mm] (bgam)
     {\textbf{Spatio-temporal BGAM}\\$s(x,y)\otimes s(\mathrm{time})$};
\node[dat, below=of bgam] (post) {Posterior of the\\condition effect};
\node[outd, text width=22mm, anchor=south west] (ppcn) at ($(post.west)+(-34mm,8mm)$) {\textbf{Posterior predictive checks:} replicated vs.\ observed data};

\draw[arr, gray!70] (p2.east) -- (p2.east -| rule.west);

\node[proc, below=of tmap] (corr) {Uncorrected, BH, BY, Holm,\\cluster mass or TFCE};
